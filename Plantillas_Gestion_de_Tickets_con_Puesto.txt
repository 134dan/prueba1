function formatNoteWithBold(raw){
  let lines = raw.split('\n');
  while(lines.length>0 && lines[lines.length-1].trim()==='') lines.pop();
  if(lines.length>=3){
    const n=lines.length;
    const firmaCompleta=lines[n-1].includes(globalGerencia);
    const cantidadFirma=firmaCompleta?3:2;
    for(let i=0;i<n;i++){
      lines[i]=i>=n-cantidadFirma?'<b>'+escapeHtml(lines[i])+'</b>':escapeHtml(lines[i]);
    }
    return lines.join('<br>');
  }
  return escapeHtml(raw).replace(/\n/g,'<br>');
}

let toastTimer=null;
function showToast(message,type){
  const toast=document.getElementById('toast');
  const messageEl=document.getElementById('toastMessage');
  const iconEl=document.getElementById('toastIcon');
  const text=String(message??'');
  let kind=type;
  if(!kind){
    if(/error|no se pudo|incorrect|fall[oó]|no autorizado|no se puede/i.test(text)) kind='error';
    else if(/no puede|debes|escribe|selecciona|vac[ií]o|advertencia/i.test(text)) kind='warning';
    else if(/copiado|guardad|actualizad|agregad|eliminad|activado|correctamente|modo consulta|firma actualizada/i.test(text)) kind='success';
    else kind='info';
  }
  const icons={success:'✓',error:'!',warning:'⚠',info:'ℹ'};
  toast.className='toast '+kind;
  messageEl.textContent=text;
  iconEl.textContent=icons[kind]||icons.info;
  toast.setAttribute('role',kind==='error'?'alert':'status');
  toast.setAttribute('aria-live',kind==='error'?'assertive':'polite');
  toast.classList.add('show');
  clearTimeout(toastTimer);
  toastTimer=setTimeout(()=>toast.classList.remove('show'),3600);
}

async function exportAllToPDF() {
  if (!isAdmin) {
    showToast('Esta función está disponible solo para administradores.', 'error');
    return;
  }

  if (typeof html2pdf === 'undefined') {
    showToast('No se pudo cargar la biblioteca para generar el PDF.', 'error');
    return;
  }

  try {
    showToast('Generando PDF, espera un momento...');

    const html = getExportDocumentHtml();
    const container = document.createElement('div');

    container.innerHTML = html;

    const contenido = container.querySelector('body') || container;
    const opciones = {
      margin: 10,
      filename: 'Plantillas_Gestion_de_Tickets.pdf',
      image: { type: 'jpeg', quality: 0.98 },
      html2canvas: { scale: 2, useCORS: true },
      jsPDF: {
        unit: 'mm',
        format: 'a4',
        orientation: 'portrait'
      },
      pagebreak: { mode: ['css', 'legacy'] }
    };

    await html2pdf().set(opciones).from(contenido).save();

    showToast('PDF descargado correctamente.', 'success');
  } catch (e) {
    showToast('No se pudo generar el PDF: ' + (e.message || e), 'error');
  }
}
